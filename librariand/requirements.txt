# librariand's dependencies, pinned.
#
# Pinned exactly because this is a deployed server, not a library: the versions
# the release workflow tested are the versions the Pi runs. install.sh rebuilds
# the venv whenever this file changes, so bumping a pin here is the whole of an
# upgrade — there is no second place to update.
#
# Installed into /usr/local/lib/librariand/venv. install.sh keeps a copy of the
# file it last installed successfully at venv/.installed-requirements and
# compares against that, so a failed install is retried rather than forgotten.

uvicorn==0.54.0
jinja2==3.1.6
python-multipart==0.0.32

# Optional at runtime — without it quarantine.py still works and only the
# tag-derived columns go blank — but the dashboard is much less useful blind.
mediafile==0.17.0

# Test-only. It is here rather than in a separate file so that
# librariand-test.sh can run on the Pi against the deployed venv, which is the
